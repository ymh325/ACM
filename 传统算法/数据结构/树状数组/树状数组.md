## 向前查 前缀和

```cpp
int query(int x){
	int res= 0;
	while(x){
		res += s[x];
		x -= lowbit(x);
	}
	return res;
}
```

## 单点增加 + 范围修改

[P3374 【模板】树状数组 1 - 洛谷](https://www.luogu.com.cn/problem/P3374)
```cpp
#include <iostream>
using namespace std;
// #define ll long long
const int N = 5e5 + 5;
int w[N];
int n, m;
int lowibit(int x){
    return x & -x;
}
void change(int pos, int k){
    while(pos <= n){
        w[pos] += k;
        pos += lowibit(pos);
    }
}
int query(int pos){
    int res = 0;
    while(pos){
        res += w[pos];
        pos -= lowibit(pos);
    }
    return res;
}
int main(){
    cin >> n >> m;
    for(int i = 1; i <= n; i++){
        int x;
        cin >> x;
        change(i, x);
    }
    for(int i = 1; i <= m; i++){
        int opt, x, y;
        cin >> opt >> x >> y;
        if(opt == 1)change(x, y);
        else cout << query(y) - query(x - 1) << endl;
    }
    return 0;
}

```

#include <iostream>

    while(pos <= n){
        w[pos] += k;
        pos += lowibit(pos);
    }
}
int query(int pos){
    int res = 0;
    while(pos){
        res += w[pos];
        pos -= lowibit(pos);

`D[]` 为差分数组。
`A[i]` 为第 `i` 位的数字，则 `A[i] = D[1]+...+D[i]`

前缀和:
$$
\begin{align}
 A_1 + A_2 +...+A_{k-1}+A_k 
 &= D_1 + (D_1+D_2) + (D_1+D_2+D_3) + ... + (D_1+D_2 +... +D_k)
 \\&=k *D_1 + (k-1)*D_2+(k-2)*D_3+...+D_k
 \\&=k * \sum^k_{i=1} D_i - \sum^k_{i=1} (i-1)*D_i
 
 \end{align}
$$

## 二维单点增加 + 范围查询

```cpp
#include <iostream>
using namespace std;
// #define ll long long
const int N = 5e4 + 5;
int w[N][N];
int n, m;
int lowibit(int x){
    return x & -x;
}
void change(int x,int y, int k){
    for(int i=x;i<=n;i += lowibit(i)){
	    for(int j=y;j<=m;j += lowibit(j)){
		    w[x][y] += k;
	    }
    }
}
int sum(int x,int y){
    int res = 0;
    for(int i=x;i<=n;i += lowibit(i)){
	    for(int j=y;j<=m;j += lowibit(j)){
		    res += w[i][j];
	    }
    }
    return res;
}
int main(){
    cin >> n >> m;
    for(int i = 1; i<=n;i++){
	    for(int j= 1;j<=m;j++){
	        int x;
	        cin >> x;
	        change(i,j, x);
	    }
    }
    return 0;
}
```
