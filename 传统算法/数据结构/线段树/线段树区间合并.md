前置知识：
- [[线段树]]

> [!NOTE] 经典模型：连续 `1` 的最长子串长度
>
> 对一个只包含 `0/1` 的序列，支持修改和查询区间内连续 `1` 的最长长度。

## 节点信息

对线段树节点区间 `[l,r]` 维护：

- `sz`：当前区间长度；
- `len`：当前区间内连续 `1` 的最长长度；
- `pre`：从区间左端点开始的连续 `1` 的最长前缀长度；
- `suf`：到区间右端点结束的连续 `1` 的最长后缀长度。

`pre` 和 `suf` 是为了处理答案跨越左右儿子的情况。只维护 `len` 无法知道左右两段拼接后能否连起来。

例如，左区间的后缀和右区间的前缀都是 `1` 时，跨越中点的连续段长度为 `left.suf + right.pre`。

## 合并两个区间

设 `L`、`R` 分别是左右儿子的统计信息，父区间为 `P`。

```text
P.sz  = L.sz + R.sz
P.len = max(L.len, R.len, L.suf + R.pre)

P.pre = L.pre
如果 L.pre == L.sz，说明左区间全是 1：
    P.pre = L.sz + R.pre

P.suf = R.suf
如果 R.suf == R.sz，说明右区间全是 1：
    P.suf = R.sz + L.suf
```

合并操作必须保留左右顺序，不能交换参数。

```cpp
Info merge(const Info& L, const Info& R) {
    Info P;
    P.sz = L.sz + R.sz;
    P.len = max({L.len, R.len, L.suf + R.pre});
    P.pre = L.pre;
    if (L.pre == L.sz) P.pre = L.sz + R.pre;
    P.suf = R.suf;
    if (R.suf == R.sz) P.suf = R.sz + L.suf;
    return P;
}
```


# 代码

[P1503 鬼子进村 - 洛谷](https://www.luogu.com.cn/problem/P1503)

> 使用set也可以解决，下面是线段树的思路

```cpp
#include <bits/stdc++.h>
using namespace std;
#define OFF ios::sync_with_stdio(0),cin.tie(0);
#define int long long
#define lc (p<<1)
#define rc (p<<1|1)
// const int mod = 1e9+7;
// const int MOD = 998244353;

const int N = 5e4 + 5;
const int MAXX = N<<2;

int n,m;
struct node{
    int l,r;
    int pre,suf;
    int len;
}tr[MAXX];
  
void up(node & res, node & ll, node & rr){
    res.len = max({ll.len,rr.len,ll.suf+rr.pre});
    res.pre = ll.pre;
    if(ll.r-ll.l+1 == ll.pre){
        res.pre = ll.pre + rr.pre;
    }
    res.suf = rr.suf;
    if(rr.r-rr.l+1 == rr.suf){
        res.suf = ll.suf + rr.suf;
    }

}

void build(int p,int l,int r){
    tr[p] = {l,r,0,0,0};
    if(l==r){
        tr[p].pre = tr[p].suf = tr[p].len = 1;
        return ;
    }
    int mid = (l+r)>>1;
    build(lc,l,mid);
    build(rc,mid+1,r);
    up(tr[p],tr[lc],tr[rc]);
}

int pos;

void update(int p,int l,int r,int op){
    if(l==r && pos == l){
        tr[p].pre = tr[p].suf = tr[p].len = (op==1);
        return ;
    }
    int mid = (l+r)>>1;
    if(pos<=mid)update(lc,l,mid,op);
    else update(rc,mid+1,r,op);

    up(tr[p],tr[lc],tr[rc]);
}

node query(int p,int l,int r,int x,int y){
    if(x<= l && r <= y){
        return tr[p];
    }
    int mid = (l+r)>>1;
    node ll, rr ;
    if (y <= mid) return query(lc, l, mid, x, y);
    if (x > mid) return query(rc, mid + 1, r, x, y);

    ll = query(lc,l,mid,x,y);
    rr = query(rc,mid+1,r,x,y);

    node res;
    res.l = ll.l;
    res.r = rr.r;

    up(res,ll,rr);

    return res;

}
signed main(){
    cin>>n>>m;
    stack<int> st;
    
    build(1,1,n);
    
    for(int i=1;i<=m;i++){
        char opt;cin>>opt;
        if(opt=='D'){
            cin>>pos;
            st.push(pos);
            update(1,1,n,0);
        }
        if(opt=='R'){
            if(st.size()){
                pos = st.top();
                st.pop();
                update(1,1,n,1);
            }
        }

        if(opt=='Q'){
            int x;
            cin>>x;
            node L = query(1,1,n,1,x);
            node R = query(1,1,n,x,n);
            int mx = L.suf + R.pre - 1;
            cout<<max(mx,0LL)<<'\n';
        }
    }
    return 0;

}
```
