> [!NOTE] 二分图的最大匹配
> 设 $G$ 为二分图，若在 $G$ 的子图 $M$ 中，任意两条边都没有公共节点，那么称 $M$ 为二分图 $G$ 的一组**匹配**。在二分图中，包含边数最多的一组匹配称为 **二分图的最大匹配**。

# 匈牙利算法


```cpp
int match[N];
int vis[N];
vector<int> e[N];

int dfs(int u){
	for(int v:e[u]){
		if(vis[v])continue;
		vis[v]=1;
		if(!match[v] || dfs(match[v])){
			match[v] = u;
			return 1;
		}
	}
	return 0;
}

signed main(){
	
	for(int i=1;i<=n;i++){
		int u,v;cin>>u>>v;
		e[u].push_back(v);
		e[v].push_back(u);
	}
	int ans = 0;
	for(int i=1;i<=n;i++){
		memset(vis,0,sizeof(vis));
		if(dfs(i))ans++;
	}
	
	return 0;
}
```
